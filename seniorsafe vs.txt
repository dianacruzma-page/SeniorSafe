import pandas as pd
from pathlib import Path
import psycopg
import sklearn
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import accuracy_score
import joblib
conexion = psycopg.connect(
    host="localhost",
    dbname="senior_safesql",
    user="postgres",
    password="1234",
    port="5432"
)
query = """
SELECT 
    a.Tipo_alerta,
    a.Nivel_riesgo,
    a.Estado,
    am.Edad,
    dc.Fase_etapa
FROM Alerta a
JOIN Hogar h ON a.Id_hogar = h.Id_hogar
JOIN Adulto_Mayor am ON h.Id_adultomayor = am.Id_adultomayor
JOIN Padecer p ON am.Id_adultomayor = p.Id_adultomayor
JOIN Diagnostico_Cognitivo dc ON p.Id_diagnostico = dc.Id_diagnostico;
"""
df = pd.read_sql(query, conexion)


df['nivel_num'] = df['nivel_riesgo'].apply(lambda x: 1 if str(x).lower() in ['critico', 'alto'] else 0)
df['fase_num'] = df['fase_etapa'].apply(lambda x: 10 if str(x).lower() in ['grave','moderada'] else 0)
total_alertas = len(df)
conteo_riesgos = df['nivel_riesgo'].value_counts()
porcentaje_riesgos = (conteo_riesgos / total_alertas) * 100

df_porcentajes = pd.DataFrame({
    'Cantidad': conteo_riesgos,
    'Porcentaje (%)': porcentaje_riesgos.round(2)
})
print("=== TABLA GENERAL COMPLETÍSIMA DE SENIOR SAFE ===")
print(df)

print("\n=== RESUMEN DE PORCENTAJES POR NIVEL DE RIESGO ===")
print(df_porcentajes)



X = df[['edad', 'fase_num']]  
Y = df['nivel_num']  

X_train, X_test, Y_train, Y_test = train_test_split(X, Y, test_size=0.2, random_state=42)

modelo = DecisionTreeClassifier()
modelo.fit(X_train, Y_train)  

print("\n¡El modelo de Inteligencia Artificial de Senior Safe ha sido entrenado con éxito!")

predicciones = modelo.predict(X_test)
precision = accuracy_score(Y_test, predicciones)
print(f"Precisión del modelo: {precision * 100}%")


adulto_nuevo = pd.DataFrame({
    'edad': [78],
    'fase_num': [10]
})

prediccion_nuevo = modelo.predict(adulto_nuevo)

if prediccion_nuevo[0] == 1:
    print("\n🚨 Alerta: El modelo predice que este adulto tiene un nivel de riesgo CRITICO O ALTO.")
else:
    print("\n✅ El modelo predice que este adulto tiene un nivel de riesgo BAJO O NORMAL.")           


joblib.dump(modelo, 'modelo_seniorsafe.pkl')
print("📦 ¡Modelo exportado con éxito a 'modelo_seniorsafe.pkl'!")


conexion.close()

from fastapi import FastAPI
from pydantic import BaseModel
import joblib
import pandas as pd

app = FastAPI()

# 1. Cargar el modelo de Machine Learning
modelo_ia = joblib.load('modelo_seniorsafe.pkl')

# 2. Definir el esquema de Pydantic
class DatosSensor(BaseModel):
    Nombre: str
    Edad: int
    Fase_num: int
    sensor_moviento: str

# 3. Definir el endpoint y la función que procesa la petición
@app.post("/evaluar_sensor")
def evaluar_sensor(datos: DatosSensor):
    
    # Crear el DataFrame correctamente con los datos recibidos
    df_ia = pd.DataFrame({
        'Edad': [datos.Edad],
        'Fase_num': [datos.Fase_num]  # Asegúrate de incluir las columnas que use tu modelo
    })
    
    # Realizar la predicción con el modelo cargado
    prediccion = modelo_ia.predict(df_ia)
    
    # Evaluar el resultado de la predicción
    if prediccion[0] == 1:
        return {
            "estado": "CRITICO",
            "accion": "Notificar a Familiar"
        }
    else:
        return {
            "estado": "NORMAL",
            "accion": "solo registrar"
        }
